            document.getElementById(`status-${i}`).textContent = `重试${retries}次成功 ✓`;
            document.getElementById(`status-${i}`).className = 'status-tag status-ok';

          testImg.onerror = function() {
            if (retries < maxRetries) {
              attempt();
            } else {

            }
          };
          testImg.src = retryUrl;
        }, 800 * retries);
      }

      attempt();

# 文件系统

Electron 提供了多种方式来访问和操作文件系统，包括 Node.js 的 `fs` 模块和 Electron 的 `dialog` 模块。

## 系统架构

## Dialog API 完整说明

### 文件对话框参数

#### showOpenDialog(options)

打开文件/目录选择对话框。

| 参数 | 类型 | 默认值 | 说明 |
|------|------|--------|------|
| `title` | String | - | 对话框标题 |
| `defaultPath` | String | - | 默认打开路径 |
| `buttonLabel` | String | - | 确认按钮文本 |
| `filters` | FileFilter[] | - | 文件类型过滤器 |
| `properties` | String[] | - | 对话框属性 |
| `message` | String | - | macOS: 对话框消息 |
| `securityScopedBookmarks` | Boolean | false | macOS: 安全范围书签 |

**properties 可选值**：

| 属性 | 说明 |
|------|------|
| `openFile` | 允许选择文件 |
| `openDirectory` | 允许选择目录 |
| `multiSelections` | 允许多选 |
| `showHiddenFiles` | 显示隐藏文件 |
| `createDirectory` | macOS: 允许创建目录 |
| `promptToCreate` | Windows: 提示创建不存在的文件 |
| `noResolveAliases` | macOS: 不解析别名 |
| `treatPackageAsDirectory` | macOS: 将包视为目录 |
| `dontAddToRecent` | Windows: 不添加到最近文件列表 |

#### showSaveDialog(options)

保存文件对话框，参数与 `showOpenDialog` 类似。

#### showMessageBox(options)

消息对话框。

| 参数 | 类型 | 默认值 | 说明 |
|------|------|--------|------|
| `type` | String | 'none' | 消息类型: 'none', 'info', 'error', 'question', 'warning' |
| `buttons` | String[] | ['OK'] | 按钮文本数组 |
| `defaultId` | Number | 0 | 默认选中按钮的索引 |
| `cancelId` | Number | 0 | 取消按钮的索引 |
| `title` | String | - | 对话框标题 |
| `message` | String | - | 消息内容 |
| `detail` | String | - | 详细信息 |
| `checkboxLabel` | String | - | 复选框文本 |
| `checkboxChecked` | Boolean | false | 复选框默认状态 |
| `icon` | NativeImage | - | 对话框图标 |
| `noLink` | Boolean | false | 不使用链接样式 |

## 文件对话框

### 打开文件

```javascript
const { dialog, app } = require('electron')

// 选择单个文件
async function openFile() {
  const result = await dialog.showOpenDialog({
    title: '选择文件',
    defaultPath: app.getPath('documents'),
    filters: [
      { name: '文本文件', extensions: ['txt', 'md'] },
      { name: '图片', extensions: ['jpg', 'png', 'gif'] },
      { name: '所有文件', extensions: ['*'] }
    ],
    properties: ['openFile']
  })
  
  if (!result.canceled && result.filePaths.length > 0) {
    return result.filePaths[0]
  }
  return null
}

// 选择多个文件
async function openFiles() {
  const result = await dialog.showOpenDialog({
    title: '选择文件',
    properties: ['openFile', 'multiSelections']
  })
  
  return result.canceled ? [] : result.filePaths
}

// 选择目录
async function selectDirectory() {
  const result = await dialog.showOpenDialog({
    title: '选择目录',
    properties: ['openDirectory', 'createDirectory']
  })
  
  if (!result.canceled && result.filePaths.length > 0) {
    return result.filePaths[0]
  }
  return null
}
```

### 保存文件

```javascript
async function saveFile(defaultName = 'untitled.txt') {
  const result = await dialog.showSaveDialog({
    title: '保存文件',
    defaultPath: defaultName,
    filters: [

      { name: 'Markdown', extensions: ['md'] },
      { name: '所有文件', extensions: ['*'] }
    ],
    properties: ['createDirectory']

  if (!result.canceled && result.filePath) {
    return result.filePath
  }
  return null
}
```

### 消息对话框

```javascript
const { dialog } = require('electron')

    checkboxChecked: false
  })

    confirmed: result.response === 1,
    dontAskAgain: result.checkboxChecked
  }
}
```

## 文件读写

### Node.js fs 模块 API

#### 常用方法

| 方法 | 同步版本 | 异步版本 | Promise 版本 | 说明 |
|------|---------|---------|-------------|------|
| 读取文件 | `readFileSync` | `readFile` | `fs.promises.readFile` | 读取文件内容 |
| 写入文件 | `writeFileSync` | `writeFile` | `fs.promises.writeFile` | 写入文件内容 |
| 追加文件 | `appendFileSync` | `appendFile` | `fs.promises.appendFile` | 追加内容到文件 |
| 创建目录 | `mkdirSync` | `mkdir` | `fs.promises.mkdir` | 创建目录 |
| 读取目录 | `readdirSync` | `readdir` | `fs.promises.readdir` | 读取目录内容 |
| 删除文件 | `unlinkSync` | `unlink` | `fs.promises.unlink` | 删除文件 |
| 删除目录 | `rmdirSync` | `rmdir` | `fs.promises.rmdir` | 删除空目录 |
| 复制文件 | `copyFileSync` | `copyFile` | `fs.promises.copyFile` | 复制文件 |
| 移动文件 | `renameSync` | `rename` | `fs.promises.rename` | 移动/重命名文件 |
| 文件状态 | `statSync` | `stat` | `fs.promises.stat` | 获取文件信息 |
| 检查存在 | `existsSync` | ❌ | ❌ | 检查文件是否存在 |

### 同步读写

```javascript
const fs = require('fs')
const path = require('path')

// 读取文件
function readFile(filePath) {
  try {
    const content = fs.readFileSync(filePath, 'utf-8')
    return { success: true, content }
  } catch (error) {
    console.error('读取文件失败:', error.message)
    return { success: false, error: error.message }
  }
}

// 写入文件
function writeFile(filePath, content) {
  try {
    fs.writeFileSync(filePath, content, 'utf-8')
    return { success: true }
  } catch (error) {
    console.error('写入文件失败:', error.message)
    return { success: false, error: error.message }
  }
}

// 读取 JSON 文件
function readJSON(filePath) {
  try {
    const content = fs.readFileSync(filePath, 'utf-8')
    return { success: true, data: JSON.parse(content) }
  } catch (error) {
    console.error('读取 JSON 失败:', error.message)
    return { success: false, error: error.message }
  }
}

// 写入 JSON 文件
function writeJSON(filePath, data, indent = 2) {
  try {
    const content = JSON.stringify(data, null, indent)
    fs.writeFileSync(filePath, content, 'utf-8')
    return { success: true }
  } catch (error) {
    console.error('写入 JSON 失败:', error.message)
    return { success: false, error: error.message }
  }
}
```

## 文件监听

### 使用 fs.watch

```javascript
const fs = require('fs')

// 监听单个文件
function watchFile(filePath, callback) {
  const watcher = fs.watch(filePath, (eventType, filename) => {
    console.log(`文件 ${filename} 发生了 ${eventType} 事件`)
    callback(eventType, filename)
  })
  
  return {
    close: () => watcher.close()
  }
}

// 监听目录
function watchDirectory(dirPath, callback) {
  const watcher = fs.watch(dirPath, { recursive: false }, (eventType, filename) => {
    console.log(`目录中的 ${filename} 发生了 ${eventType} 事件`)
    callback(eventType, filename)
  })
  
  return {
    close: () => watcher.close()
  }
}
```

### 使用 chokidar（推荐）

[chokidar](https://github.com/paulmillr/chokidar) 是一个跨平台的文件监听库，解决了 `fs.watch` 的许多问题。

```bash
npm install chokidar
```

```javascript
const chokidar = require('chokidar')

// 监听文件或目录
function watch(path, options = {}) {
  const defaultOptions = {
    ignored: /(^|[\/\\])\../,  // 忽略点文件
    persistent: true,
    ignoreInitial: false,      // 是否忽略初始扫描
    awaitWriteFinish: {
      stabilityThreshold: 2000,  // 文件稳定时间（毫秒）
      pollInterval: 100          // 轮询间隔
    }
  }
  
  const watcher = chokidar.watch(path, { ...defaultOptions, ...options })
  
  watcher
    .on('add', path => console.log(`文件添加: ${path}`))
    .on('change', path => console.log(`文件修改: ${path}`))
    .on('unlink', path => console.log(`文件删除: ${path}`))
    .on('addDir', path => console.log(`目录添加: ${path}`))
    .on('unlinkDir', path => console.log(`目录删除: ${path}`))
    .on('error', error => console.log(`监听错误: ${error}`))
    .on('ready', () => console.log('初始扫描完成，开始监听'))
  
  return watcher
}

// 监听多个路径
const watcher = chokidar.watch([
  '/path/to/file1',
  '/path/to/file2',
  '/path/to/directory'
])

// 取消监听
watcher.close()
```

### 复制/移动文件

```javascript
const fs = require('fs')

// 复制文件
function copyFile(src, dest) {
  try {
    // 确保目标目录存在
    const destDir = path.dirname(dest)
    if (!fs.existsSync(destDir)) {
      fs.mkdirSync(destDir, { recursive: true })
    }
    
    fs.copyFileSync(src, dest)
    return { success: true }
  } catch (error) {
    return { success: false, error: error.message }
  }
}

// 复制目录（递归）
function copyDirectory(src, dest) {
  try {
    fs.cpSync(src, dest, { recursive: true })
    return { success: true }
  } catch (error) {
    return { success: false, error: error.message }
  }
}

// 移动/重命名文件
function moveFile(oldPath, newPath) {
  try {
    // 确保目标目录存在
    const newDir = path.dirname(newPath)
    if (!fs.existsSync(newDir)) {
      fs.mkdirSync(newDir, { recursive: true })
    }
    
    fs.renameSync(oldPath, newPath)
    return { success: true }
  } catch (error) {
    return { success: false, error: error.message }
  }
}
```

### 获取文件信息

```javascript
const fs = require('fs')

function getFileInfo(filePath) {
  try {
    const stats = fs.statSync(filePath)
    
    return {
      success: true,
      info: {
        exists: true,
        isFile: stats.isFile(),
        isDirectory: stats.isDirectory(),
        size: stats.size,
        sizeFormatted: formatFileSize(stats.size),
        created: stats.birthtime,
        modified: stats.mtime,
        accessed: stats.atime,
        permissions: stats.mode,
        owner: stats.uid,
        group: stats.gid
      }
    }
  } catch (error) {
    if (error.code === 'ENOENT') {
      return { success: true, info: { exists: false } }
    }
    return { success: false, error: error.message }
  }
}

// 格式化文件大小
function formatFileSize(bytes) {
  if (bytes === 0) return '0 Bytes'
  const k = 1024
  const sizes = ['Bytes', 'KB', 'MB', 'GB', 'TB']
  const i = Math.floor(Math.log(bytes) / Math.log(k))
  return Math.round(bytes / Math.pow(k, i) * 100) / 100 + ' ' + sizes[i]
}
```

## 遍历目录

### 同步遍历

```javascript
const fs = require('fs')
const path = require('path')

// 遍历目录（深度优先）
function walkDir(dir, callback) {
  try {
    const files = fs.readdirSync(dir)
    
    for (const file of files) {
      const filePath = path.join(dir, file)
      const stats = fs.statSync(filePath)
      
      if (stats.isDirectory()) {
        walkDir(filePath, callback)
      } else {
        callback(filePath, stats)
      }
    }
  } catch (error) {
    console.error(`遍历目录失败: ${dir}`, error.message)
  }
}

// 获取所有文件列表
function getAllFiles(dir, fileList = []) {
  const files = fs.readdirSync(dir)
  
  for (const file of files) {
    const filePath = path.join(dir, file)
    const stats = fs.statSync(filePath)
    
    if (stats.isDirectory()) {
      getAllFiles(filePath, fileList)
    } else {
      fileList.push({
        path: filePath,
        name: file,
        size: stats.size,
        modified: stats.mtime
      })
    }
  }
  
  return fileList
}
```

### 异步遍历（推荐）

```javascript
const fs = require('fs/promises')
const path = require('path')

// 异步遍历目录
async function walkDirAsync(dir, callback) {
  try {
    const files = await fs.readdir(dir)
    
    for (const file of files) {
      const filePath = path.join(dir, file)
      const stats = await fs.stat(filePath)
      
      if (stats.isDirectory()) {
        await walkDirAsync(filePath, callback)
      } else {
        await callback(filePath, stats)
      }
    }
  } catch (error) {
    console.error(`遍历目录失败: ${dir}`, error.message)
  }
}

// 获取所有文件列表（异步）
async function getAllFilesAsync(dir) {
  const results = []
  const files = await fs.readdir(dir)
  
  for (const file of files) {
    const filePath = path.join(dir, file)
    const stats = await fs.stat(filePath)
    
    if (stats.isDirectory()) {
      const subResults = await getAllFilesAsync(filePath)
      results.push(...subResults)
    } else {
      results.push({
        path: filePath,
        name: file,
        size: stats.size,
        modified: stats.mtime
      })
    }
  }
  
  return results
}
```

## 路径处理

### path 模块 API

```javascript
const path = require('path')

// 路径拼接
const fullPath = path.join('/foo', 'bar', 'baz.txt')
// 结果: '/foo/bar/baz.txt'

// 解析绝对路径
const resolved = path.resolve('foo', 'bar', 'baz.txt')
// 结果: '/当前工作目录/foo/bar/baz.txt'

// 获取目录名
const dir = path.dirname('/foo/bar/baz.txt')
// 结果: '/foo/bar'

// 获取文件名
const base = path.basename('/foo/bar/baz.txt')
// 结果: 'baz.txt'

// 获取文件名（不含扩展名）
const name = path.basename('/foo/bar/baz.txt', '.txt')
// 结果: 'baz'

// 获取扩展名
const ext = path.extname('/foo/bar/baz.txt')
// 结果: '.txt'

// 解析路径
const parsed = path.parse('/foo/bar/baz.txt')
// 结果: { root: '/', dir: '/foo/bar', base: 'baz.txt', ext: '.txt', name: 'baz' }

// 格式化路径
const formatted = path.format({
  dir: '/foo/bar',
  base: 'baz.txt'
})
// 结果: '/foo/bar/baz.txt'

// 规范化路径
const normalized = path.normalize('/foo/bar/../baz/./qux')
// 结果: '/foo/baz/qux'

// 相对路径
const relative = path.relative('/foo/bar', '/foo/baz/qux')
// 结果: '../baz/qux'
```

## 大文件处理

### 流式读写

```javascript
const fs = require('fs')
const path = require('path')

// 大文件复制（使用流）
function copyLargeFile(src, dest, onProgress) {
  return new Promise((resolve, reject) => {
    const readStream = fs.createReadStream(src)
    const writeStream = fs.createWriteStream(dest)
    let copied = 0
    const total = fs.statSync(src).size
    
    readStream.on('data', (chunk) => {
      copied += chunk.length
      if (onProgress) {
        onProgress(copied, total)
      }
    })
    
    readStream.on('error', reject)
    writeStream.on('error', reject)
    writeStream.on('finish', resolve)
    
    readStream.pipe(writeStream)
  })
}

// 分块读取大文件
function readLargeFileInChunks(filePath, chunkSize = 1024 * 1024) {
  const stats = fs.statSync(filePath)
  const fileSize = stats.size
  const chunks = []
  
  for (let offset = 0; offset < fileSize; offset += chunkSize) {
    const fd = fs.openSync(filePath, 'r')
    const buffer = Buffer.alloc(Math.min(chunkSize, fileSize - offset))
    fs.readSync(fd, buffer, 0, buffer.length, offset)
    fs.closeSync(fd)
    chunks.push(buffer)
  }
  
  return Buffer.concat(chunks)
}

// 行读取器（逐行读取大文件）
function readLines(filePath, callback) {
  return new Promise((resolve, reject) => {
    const readStream = fs.createReadStream(filePath, { encoding: 'utf-8' })
    const readline = require('readline')
    
    const rl = readline.createInterface({
      input: readStream,
      crlfDelay: Infinity
    })
    
    rl.on('line', (line) => {
      callback(line)
    })
    
    rl.on('close', resolve)
    rl.on('error', reject)
  })
}
```

```javascript
async function searchFiles(dir, pattern) {
  const results = []
  const files = await fs.promises.readdir(dir)
  
  for (const file of files) {
    const filePath = path.join(dir, file)
    const stats = await fs.promises.stat(filePath)
    
    if (stats.isDirectory()) {
      const subResults = await searchFiles(filePath, pattern)
      results.push(...subResults)
    } else if (file.includes(pattern)) {
      results.push(filePath)
    }
  }
  
  return results
```
